# Galvanic tree (plasm). Anglophone countries favour it, Muslim countries shun it.
vatmoss_grafting = {
	era = era_2
	texture = "gfx/interface/icons/invention_icons/pharmaceuticals.dds"
	category = military

	ai_weight = {
		value = 1
		if = {
			limit = { mfk_is_anglophone_country = yes }
			multiply = 4
		}
		if = {
			limit = { mfk_is_muslim_country = yes }
			multiply = 0.25
		}
	}
}
