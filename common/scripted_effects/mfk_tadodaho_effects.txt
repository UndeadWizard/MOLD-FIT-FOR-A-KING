mfk_tadodaho_create = {
	create_character = {
		female = no # the title of Tadodaho is held by men
		culture = cu:iroquoian
		religion = rel:animist
		interest_group = ig_devout
		ideology = ideology_traditionalist
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_tadodaho
	}
	if = {
		limit = { exists = scope:new_tadodaho }
		scope:new_tadodaho = {
			add_character_role = character_role_tadodaho
			change_character_religion = rel:animist
		}
		set_variable = {
			name = tadodaho_var
			value = scope:new_tadodaho
		}
		if = {
			limit = { NOT = { has_modifier = mfk_onondaga_council_fire } }
			add_modifier = { name = mfk_onondaga_council_fire }
		}
	}
}

# Make sure the Iroquois have a Tadodaho (game start, or if the office has emptied)
mfk_tadodaho_ensure = {
	c:IRO ?= {
		if = {
			limit = {
				NOT = { has_variable = mfk_council_fire_extinguished }
				NOT = { any_scope_character = { has_role = character_role_tadodaho } }
			}
			if = { # 1836: Samuel George is the Tadodaho
				limit = {
					NOT = { has_variable = mfk_samuel_george_raised }
					any_scope_character = { has_template = IRO_samuel_george }
				}
				set_variable = mfk_samuel_george_raised
				random_scope_character = {
					limit = { has_template = IRO_samuel_george }
					save_scope_as = new_tadodaho
					add_character_role = character_role_tadodaho
				}
				set_variable = {
					name = tadodaho_var
					value = scope:new_tadodaho
				}
				if = {
					limit = { NOT = { has_modifier = mfk_onondaga_council_fire } }
					add_modifier = { name = mfk_onondaga_council_fire }
				}
			}
			else = {
				mfk_tadodaho_create = yes
			}
		}
		if = {
			limit = {
				NOT = { has_journal_entry = je_tadodaho }
				NOT = { has_variable = mfk_council_fire_concluded }
			}
			add_journal_entry = { type = je_tadodaho }
		}
		# The Tadodaho keeps the Longhouse religion. Joining the Clan Mothers converts members to the state religion, so restore it.
		every_scope_character = {
			limit = {
				has_role = character_role_tadodaho
				NOT = { religion = rel:animist }
			}
			change_character_religion = rel:animist
		}
		# While the Tadodaho leads the Confederacy, each new Tadodaho also becomes its leader
		if = {
			limit = {
				has_variable = mfk_tadodaho_rules
				NOT = { ruler ?= { has_role = character_role_tadodaho } }
				any_scope_character = { has_role = character_role_tadodaho }
			}
			random_scope_character = {
				limit = { has_role = character_role_tadodaho }
				set_character_as_ruler = yes
			}
		}
		# Keep the Horns of John Brant widget pointed at whoever is Grand Hoyaneh now
		if = {
			limit = {
				exists = ruler
				NOT = { var:grand_hoyaneh_var ?= ruler }
			}
			set_variable = {
				name = grand_hoyaneh_var
				value = ruler
			}
		}
		mfk_council_harmony_update = yes
		mfk_clan_mothers_ig_update = yes
		mfk_sheltering_nations_update = yes
		mfk_great_law_nations_setup = yes
		mfk_council_weights_update = yes
		if = {
			limit = {
				has_journal_entry = je_tadodaho
				NOT = { any_political_movement = { is_political_movement_type = movement_great_law } }
			}
			create_political_movement = { type = movement_great_law }
		}
		if = {
			limit = {
				has_journal_entry = je_tadodaho
				NOT = { any_political_movement = { is_political_movement_type = movement_dehorners } }
			}
			create_political_movement = { type = movement_dehorners }
		}
		mfk_sheltered_departure_check = yes
		if = { # GDP goal for the Horns of John Brant, set the same way as Japan's Tenpo goal
			limit = { NOT = { has_variable = mfk_brant_gdp_goal } }
			set_variable = {
				name = mfk_brant_gdp_goal
				value = {
					value = gdp
					multiply = 1.35
				}
			}
		}
		mfk_shelter_request_check = yes
	}
}

mfk_clan_mothers_ig_update = {
	ig:ig_devout ?= {
		save_scope_as = clan_mothers_ig
		set_interest_group_name = ig_clan_mothers
	}
	ig:ig_landowners ?= { set_interest_group_name = ig_royaneh }
	ig:ig_rural_folk ?= { set_interest_group_name = ig_longhouse_families }
	if = {
		limit = {
			exists = scope:clan_mothers_ig
			scope:clan_mothers_ig = {
				OR = {
					NOT = { exists = leader }
					leader ?= { is_female = no }
				}
			}
		}
		if = {
			limit = {
				any_scope_character = {
					interest_group ?= scope:clan_mothers_ig
					is_female = yes
					is_ruler_of_own_country = no
					is_heir_of_own_country = no
				}
			}
			random_scope_character = {
				limit = {
					interest_group ?= scope:clan_mothers_ig
					is_female = yes
					is_ruler_of_own_country = no
					is_heir_of_own_country = no
				}
				set_as_interest_group_leader = yes
			}
		}
		else = {
			create_character = {
				female = yes
				culture = cu:iroquoian
				interest_group = ig_devout
				ideology = ideology_traditionalist
				home_region = STATE_SIX_NATIONS
				ig_leader = yes
			}
		}
	}
}

mfk_council_harmony_update = {
	if = {
		limit = { je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 70 } }
		if = {
			limit = { NOT = { has_modifier = mfk_council_harmony_high } }
			add_modifier = { name = mfk_council_harmony_high }
		}
	}
	else_if = {
		limit = { has_modifier = mfk_council_harmony_high }
		remove_modifier = mfk_council_harmony_high
	}
	if = {
		limit = { je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" <= 30 } }
		if = {
			limit = { NOT = { has_modifier = mfk_council_harmony_low } }
			add_modifier = { name = mfk_council_harmony_low }
		}
	}
	else_if = {
		limit = { has_modifier = mfk_council_harmony_low }
		remove_modifier = mfk_council_harmony_low
	}
	if = {
		limit = {
			OR = {
				je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" <= 15 }
				AND = {
					has_law = law_type:law_council_fires
					je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" <= 25 }
				}
			}
			NOT = { has_journal_entry = je_dehorning_crisis }
			NOT = { has_variable = mfk_tadodaho_rules }
			NOT = { has_variable = mfk_dehorning_cooldown }
			exists = ruler
		}
		set_variable = {
			name = mfk_dehorning_target
			value = ruler
		}
		add_journal_entry = { type = je_dehorning_crisis }
		trigger_event = { id = mfk_tadodaho.3 popup = yes }
	}
}

# Change Council Harmony by $VALUE$.
mfk_change_council_harmony = {
	je:je_tadodaho ?= {
		add_progress = { value = $VALUE$ name = mfk_council_harmony_bar }
	}
}

mfk_dehorning_cleanup = {
	remove_variable = mfk_dehorning_target
	set_variable = {
		name = mfk_dehorning_cooldown
		years = 5
	}
}

# The Onondaga Council Fire ends with the Great Law renewed. Scope = Iroquois
mfk_great_law_renewed_effect = {
	set_variable = mfk_council_fire_concluded
	add_modifier = { name = mfk_great_law_renewed }
	trigger_event = { id = mfk_tadodaho.9 popup = yes }
}

# The Onondaga Council Fire ends with the fire put out: the Tadodaho's office is abolished. Scope = Iroquois
mfk_council_fire_extinguished_effect = {
	set_variable = mfk_council_fire_concluded
	set_variable = mfk_council_fire_extinguished
	custom_tooltip = mfk_fire_out_effect_tt
	hidden_effect = {
		every_scope_character = {
			limit = { has_role = character_role_tadodaho }
			remove_character_role = character_role_tadodaho
		}
		remove_variable = tadodaho_var
		remove_variable = mfk_tadodaho_rules
		if = {
			limit = { has_modifier = mfk_onondaga_council_fire }
			remove_modifier = mfk_onondaga_council_fire
		}
		if = {
			limit = { has_law = law_type:law_council_fires }
			activate_law = law_type:law_parliamentary_republic
		}
	}
	add_modifier = { name = mfk_council_fire_extinguished }
	trigger_event = { id = mfk_tadodaho.10 popup = yes }
}
