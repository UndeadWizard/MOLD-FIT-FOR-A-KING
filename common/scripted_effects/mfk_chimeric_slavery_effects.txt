# Root = country. Fired from on_yearly_pulse_country.
# Chimeric Slavery blends Legacy and Debt Slavery:
#  Legacy: existing slaves remain slaves and are refined into Verdigri.
#  Debt:   free Verdigri are bound only when poor, for the cost of their
#          refinement. Wealthier and middle/upper strata Verdigri stay free,
#          and binding stops once a state is a quarter slaves.
#  Penal:  criminals, orphans and the destitute are sentenced to the vats
#          and leave as free but penniless Verdigri, open to binding.
# Refinement and penal sentences only run in states whose market sells plasm.
mfk_chimeric_slavery_yearly_effect = {
	if = {
		limit = { has_law = law_type:law_chimeric_slavery }

		every_scope_state = {
			# Debt binding runs first, so last year's penal converts are exposed to it
			if = {
				limit = {
					pop_type_percent_state = {
						target = pop_type:slaves
						value < 0.25
					}
				}
				every_scope_pop = {
					limit = {
						culture = cu:verdigri
						strata = lower
						NOT = { is_pop_type = slaves }
						wealth <= 10
					}
					if = {
						limit = { wealth <= 5 }
						random = {
							chance = {
							value = 50
							if = {
								limit = { has_global_variable = mfk_cotton_collapse }
								add = 10
							}
						}
							change_poptype = pop_type:slaves
						}
					}
					else = {
						random = {
							chance = {
							value = 20
							if = {
								limit = { has_global_variable = mfk_cotton_collapse }
								add = 10
							}
						}
							change_poptype = pop_type:slaves
						}
					}
				}
			}

			if = {
				limit = {
					market ?= {
						mg:plasm = {
							market_goods_sell_orders > 0
						}
					}
				}

				# Planters refine their slaves
				every_scope_pop = {
					limit = {
						is_pop_type = slaves
						NOT = { culture = cu:verdigri }
					}
					change_pop_culture = {
						target = cu:verdigri
						value = 0.03
					}
				}

				# Penal refinement needs police to make the arrests
				if = {
					limit = {
						owner = {
							NOT = { has_law = law_type:law_no_police }
						}
					}
					every_scope_pop = {
						limit = {
							strata = lower
							is_employed = no
							NOT = { is_pop_type = slaves }
							NOT = { culture = cu:verdigri }
						}
						if = {
							# Orphanages sell children to the vats where child labour is legal
							limit = {
								owner = { has_law = law_type:law_child_labor_allowed }
							}
							change_pop_culture = {
								target = cu:verdigri
								value = 0.02
							}
						}
						else = {
							change_pop_culture = {
								target = cu:verdigri
								value = 0.01
							}
						}
					}
				}
			}
		}
	}
}

# Root = country. Fired from on_yearly_pulse_country.
# The Chimeric Question: in the NAU's Slave States, poor free Verdigri are bound into slavery,
# unless the state is under the Verdigri Crackdown decree.
mfk_chimeric_question_yearly_effect = {
	if = {
		limit = { has_law = law_type:law_chimeric_question }
		every_scope_state = {
			limit = {
				is_slave_state = yes
				NOT = { has_decree = decree_mfk_verdigri_crackdown }
				pop_type_percent_state = {
					target = pop_type:slaves
					value < 0.25
				}
			}
			every_scope_pop = {
				limit = {
					culture = cu:verdigri
					strata = lower
					NOT = { is_pop_type = slaves }
					wealth <= 10
				}
				if = {
					limit = { wealth <= 5 }
					random = {
						chance = {
							value = 25
							if = {
								limit = { has_global_variable = mfk_cotton_collapse }
								add = 10
							}
						}
						change_poptype = pop_type:slaves
					}
				}
				else = {
					random = {
						chance = {
							value = 10
							if = {
								limit = { has_global_variable = mfk_cotton_collapse }
								add = 10
							}
						}
						change_poptype = pop_type:slaves
					}
				}
			}
		}
		# Back-street vats: unlicensed refiners turn the destitute into free Verdigri, ripe for binding next year
		every_scope_state = {
			limit = {
				is_slave_state = yes
				NOT = { has_decree = decree_mfk_verdigri_crackdown }
				market ?= {
					mg:plasm = {
						market_goods_sell_orders > 0
					}
				}
			}
			every_scope_pop = {
				limit = {
					strata = lower
					is_employed = no
					NOT = { is_pop_type = slaves }
					NOT = { culture = cu:verdigri }
				}
				change_pop_culture = {
					target = cu:verdigri
					value = 0.005
				}
			}
		}
	}
}

# Root = the NAU. The Verdigri Compromise: Verdigri slavery is made law across the Union, in exchange for
# the Governorships holding no unrefined man as property (they refine their slaves instead).
mfk_verdigri_compromise_effect = {
	if = {
		limit = {
			NOT = { has_variable = mfk_verdigri_compromise }
			NOT = { has_variable = mfk_calhoun_revolt_started }
		}
		if = {
			limit = { NOT = { has_law = law_type:law_chimeric_slavery } }
			activate_law = law_type:law_chimeric_slavery
		}
		every_subject_or_below = {
			limit = {
				OR = {
					c:SCR ?= THIS
					c:NCR ?= THIS
					c:GRG ?= THIS
					c:FLR ?= THIS
				}
				NOT = { has_law = law_type:law_chimeric_slavery }
			}
			activate_law = law_type:law_chimeric_slavery
		}
		set_variable = mfk_verdigri_compromise
		if = {
			limit = {
				has_variable = mfk_cq_question_var
				var:mfk_cq_question_var >= 75
			}
			add_modifier = { name = mfk_compromise_defiance }
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_slavery_banned
					value > neutral
				}
			}
			add_modifier = {
				name = delayed_emancipation
				days = normal_modifier_time
			}
		}
		c:GBR ?= {
			change_relations = {
				country = ROOT
				value = -20
			}
		}	}

}

# Root = the NAU. Enforce the Act: abolition everywhere, and the governorships rise as the Cavalier Confederation.
mfk_enforce_abolition_effect = {
	if = {
		limit = {
			NOT = { has_variable = mfk_verdigri_compromise }
			NOT = { has_variable = mfk_calhoun_revolt_started }
		}
		if = {
			limit = { NOT = { has_law = law_type:law_slavery_banned } }
			activate_law = law_type:law_slavery_banned
		}
		set_variable = mfk_calhoun_revolt_started
		save_scope_as = mfk_union_scope
		if = {
			limit = {
				exists = c:SCR
				c:SCR = { is_subject_of = ROOT }
			}
			c:SCR = {
				save_scope_as = mfk_rebel_scope
				if = {
					limit = {
						exists = c:NCR
						c:NCR = {
							is_subject_of = ROOT
							is_player = no
							OR = {
								NOT = { ROOT = { has_variable = mfk_cq_compensation } }
								liberty_desire >= 50
							}
						}
					}
					annex = c:NCR
				}
				if = {
					limit = {
						exists = c:GRG
						c:GRG = {
							is_subject_of = ROOT
							is_player = no
							OR = {
								NOT = { ROOT = { has_variable = mfk_cq_compensation } }
								liberty_desire >= 50
							}
						}
					}
					annex = c:GRG
				}
				if = {
					limit = {
						exists = c:FLR
						c:FLR = {
							is_subject_of = ROOT
							is_player = no
							OR = {
								NOT = { ROOT = { has_variable = mfk_cq_compensation } }
								liberty_desire >= 50
							}
						}
					}
					annex = c:FLR
				}
				change_tag = CAV
				create_diplomatic_play = {
					target_country = scope:mfk_union_scope
					type = dp_independence
					add_war_goal = {
						holder = scope:mfk_union_scope
						target_country = scope:mfk_rebel_scope
						type = annex_country
					}
				}
			}
		}
		trigger_event = { id = mfk_calhoun.1 popup = yes }	}

}

# Root = the NAU. A Chimeric Question bill is off the floor: clear it and put a year between bills.
mfk_cq_clear_bill_effect = {
	remove_variable = mfk_cq_bill
	if = {
		limit = { has_variable = mfk_cq_bill_compromise }
		remove_variable = mfk_cq_bill_compromise
	}
	if = {
		limit = { has_variable = mfk_cq_bill_enforce }
		remove_variable = mfk_cq_bill_enforce
	}
	if = {
		limit = { has_variable = mfk_cq_vote_called }
		remove_variable = mfk_cq_vote_called
	}
	set_variable = {
		name = mfk_cq_bill_cooldown
		days = 365
	}
}

# Root = the NAU. The bill carries: end the enactment and settle the Chimeric Question its way.
mfk_cq_pass_bill_effect = {
	# Ends the enactment process quietly: the law itself is activated below
	if = {
		limit = {
			OR = {
				is_enacting_law = law_type:law_chimeric_slavery
				is_enacting_law = law_type:law_slavery_banned
			}
		}
		hidden_effect = {
			cancel_enactment = yes
		}
	}
	if = {
		limit = { has_variable = mfk_cq_bill_compromise }
		mfk_verdigri_compromise_effect = yes
	}
	else = {
		mfk_enforce_abolition_effect = yes
	}
	mfk_cq_clear_bill_effect = yes
}
