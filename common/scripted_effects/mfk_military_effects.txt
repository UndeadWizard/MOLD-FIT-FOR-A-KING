# Root = country. Fired from on_monthly_pulse_country.

# Sets the formation-ratio traits on each army general:
#  Beasts: up to 30% of the formation is Beastmasters, more and they Stampede.
#  Constructs: up to 40% is a Construct Core, more and they lack officers to direct them.
mfk_formation_composition_update = {
	every_scope_character = {
		limit = {
			has_role_of_type = general
			has_variable = mfk_formation_checked
		}
		remove_variable = mfk_formation_checked
	}

	every_military_formation = {
		limit = { is_army = yes }
		save_scope_as = mfk_formation

		every_scope_general = {
			set_variable = mfk_formation_checked

			# Beasts
			if = {
				limit = { scope:mfk_formation = { mfk_formation_stampede = yes } }
				if = {
					limit = { NOT = { has_trait = mfk_stampede } }
					add_trait = mfk_stampede
				}
			}
			else_if = {
				limit = { scope:mfk_formation = { mfk_formation_beastmasters = yes } }
				if = {
					limit = { NOT = { has_trait = mfk_beastmasters } }
					add_trait = mfk_beastmasters
				}
			}
			else = {
				mfk_remove_beast_traits = yes
			}

			# Constructs
			if = {
				limit = { scope:mfk_formation = { mfk_formation_unguided_constructs = yes } }
				if = {
					limit = { NOT = { has_trait = mfk_unguided_constructs } }
					add_trait = mfk_unguided_constructs
				}
			}
			else_if = {
				limit = { scope:mfk_formation = { mfk_formation_construct_core = yes } }
				if = {
					limit = { NOT = { has_trait = mfk_construct_core } }
					add_trait = mfk_construct_core
				}
			}
			else = {
				mfk_remove_construct_traits = yes
			}
		}
	}

	# Generals no longer commanding an army formation lose the traits
	every_scope_character = {
		limit = {
			NOT = { has_variable = mfk_formation_checked }
			OR = {
				has_trait = mfk_beastmasters
				has_trait = mfk_stampede
				has_trait = mfk_construct_core
				has_trait = mfk_unguided_constructs
			}
		}
		mfk_remove_beast_traits = yes
		mfk_remove_construct_traits = yes
	}
}

# Character scope.
mfk_remove_beast_traits = {
	if = {
		limit = { has_trait = mfk_beastmasters }
		remove_trait = mfk_beastmasters
	}
	if = {
		limit = { has_trait = mfk_stampede }
		remove_trait = mfk_stampede
	}
}

mfk_remove_construct_traits = {
	if = {
		limit = { has_trait = mfk_construct_core }
		remove_trait = mfk_construct_core
	}
	if = {
		limit = { has_trait = mfk_unguided_constructs }
		remove_trait = mfk_unguided_constructs
	}
}

# Political and supply costs of leaning on exotic troops:
#  Galvanic 40%+: war support falls faster with casualties and the Devout object.
#  Voltaic 40%+: Devout in Muslim countries approve of the inscribed ranks,
#                Devout elsewhere object to the constructs as idols.
#  Units supplied wholly by plasm or iksir 40%+: military goods cost more.
mfk_army_composition_update = {
	# Galvanic
	if = {
		limit = { mfk_army_galvanic_share >= 0.4 }
		if = {
			limit = { NOT = { has_modifier = mfk_vat_legions } }
			add_modifier = { name = mfk_vat_legions }
		}
		ig:ig_devout ?= {
			if = {
				limit = { NOT = { has_modifier = mfk_galvanic_abomination } }
				add_modifier = { name = mfk_galvanic_abomination }
			}
		}
	}
	else = {
		if = {
			limit = { has_modifier = mfk_vat_legions }
			remove_modifier = mfk_vat_legions
		}
		ig:ig_devout ?= {
			if = {
				limit = { has_modifier = mfk_galvanic_abomination }
				remove_modifier = mfk_galvanic_abomination
			}
		}
	}

	# Voltaic
	if = {
		limit = {
			mfk_army_voltaic_share >= 0.4
			mfk_is_muslim_country = yes
		}
		ig:ig_devout ?= {
			if = {
				limit = { has_modifier = mfk_heathen_idols }
				remove_modifier = mfk_heathen_idols
			}
			if = {
				limit = { NOT = { has_modifier = mfk_inscribed_ranks } }
				add_modifier = { name = mfk_inscribed_ranks }
			}
		}
	}
	else_if = {
		limit = { mfk_army_voltaic_share >= 0.4 }
		ig:ig_devout ?= {
			if = {
				limit = { has_modifier = mfk_inscribed_ranks }
				remove_modifier = mfk_inscribed_ranks
			}
			if = {
				limit = { NOT = { has_modifier = mfk_heathen_idols } }
				add_modifier = { name = mfk_heathen_idols }
			}
		}
	}
	else = {
		ig:ig_devout ?= {
			if = {
				limit = { has_modifier = mfk_inscribed_ranks }
				remove_modifier = mfk_inscribed_ranks
			}
			if = {
				limit = { has_modifier = mfk_heathen_idols }
				remove_modifier = mfk_heathen_idols
			}
		}
	}

	# Supply
	if = {
		limit = { mfk_army_exotic_supplied_share >= 0.4 }
		if = {
			limit = { NOT = { has_modifier = mfk_exotic_supply_strain } }
			add_modifier = { name = mfk_exotic_supply_strain }
		}
	}
	else_if = {
		limit = { has_modifier = mfk_exotic_supply_strain }
		remove_modifier = mfk_exotic_supply_strain
	}
}
