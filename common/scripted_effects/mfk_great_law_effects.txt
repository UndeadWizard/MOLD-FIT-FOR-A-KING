# The nations of the Great Law of Peace, and the unaligned nations who have settled among them.
# Scope = Iroquois.
mfk_create_nation_chief = {
	create_character = {
		female = no
		culture = cu:$CULTURE$
		religion = rel:$RELIGION$
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
		on_created = {
			set_variable = { name = mfk_nation value = flag:$NATION$ }
		}
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}

mfk_replace_nation_chief = {
	save_scope_as = dead_nation_chief
	owner ?= {
		if = {
			limit = { is_target_in_variable_list = { name = mfk_great_law_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_great_law_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_great_law_list }
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = mfk_unaligned_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_unaligned_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_unaligned_list }
		}
	}
}

mfk_nation_successor = {
	mfk_roll_chief_$LIST$ = yes
}

# Creates the successor with a given interest group and ideology. Scope = Iroquois.
mfk_nation_successor_create = {
	create_character = {
		female = no
		culture = scope:dead_nation_chief.culture
		religion = scope:dead_nation_chief.religion
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
		set_variable = { name = mfk_nation value = scope:dead_nation_chief.var:mfk_nation }
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}

# Saves made before the sheltering nations were reworked: retire the old chiefs and raise the new ones. Scope = Iroquois.
mfk_sheltering_nations_update = {
	if = {
		limit = {
			has_variable = mfk_nations_setup
			NOT = { has_variable = mfk_sheltering_nations_v2 }
		}
		set_variable = mfk_sheltering_nations_v2
		every_in_list = {
			variable = mfk_unaligned_list
			remove_character_role = character_role_nation_chief
			retire_character = yes
		}
		clear_variable_list = mfk_unaligned_list
		mfk_create_nation_chief = { NATION = ohionon IG = ig_armed_forces CULTURE = algonquian RELIGION = animist LIST = mfk_unaligned_list IDEOLOGY = traditionalist }
		mfk_create_nation_chief = { NATION = erie_freemen IG = ig_trade_unions CULTURE = afro_american RELIGION = protestant LIST = mfk_unaligned_list IDEOLOGY = abolitionist }
		mfk_create_nation_chief = { NATION = niagara_metis IG = ig_petty_bourgeoisie CULTURE = metis RELIGION = catholic LIST = mfk_unaligned_list IDEOLOGY = moderate }
		mfk_create_nation_chief = { NATION = wabash_shawnee IG = ig_armed_forces CULTURE = algonquian RELIGION = animist LIST = mfk_unaligned_list IDEOLOGY = jingoist_leader }
	}
	else_if = {
		limit = { NOT = { has_variable = mfk_sheltering_nations_v2 } }
		set_variable = mfk_sheltering_nations_v2
	}
}
