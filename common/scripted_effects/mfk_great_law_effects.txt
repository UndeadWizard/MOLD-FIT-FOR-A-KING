# The nations of the Great Law of Peace, and the unaligned nations who have settled among them.
# Scope = Iroquois.
mfk_create_nation_chief = {
	create_character = {
		female = no
		culture = cu:$CULTURE$
		religion = rel:$RELIGION$
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
		on_created = {
			set_variable = { name = mfk_nation value = flag:$NATION$ }
		}
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
		change_character_religion = rel:$RELIGION$
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}

mfk_replace_nation_chief = {
	save_scope_as = dead_nation_chief
	owner ?= {
		if = {
			limit = { is_target_in_variable_list = { name = mfk_great_law_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_great_law_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_great_law_list }
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = mfk_unaligned_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_unaligned_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_unaligned_list }
		}
	}
}

mfk_nation_successor = {
	mfk_roll_chief_$LIST$ = yes
}

# Creates the successor with a given interest group and ideology. Scope = Iroquois.
mfk_nation_successor_create = {
	create_character = {
		female = no
		culture = scope:dead_nation_chief.culture
		religion = scope:dead_nation_chief.religion
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
		change_character_religion = scope:dead_nation_chief.religion
		set_variable = { name = mfk_nation value = scope:dead_nation_chief.var:mfk_nation }
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}

# Saves made before the sheltering nations were reworked: retire the old chiefs and raise the new ones. Scope = Iroquois.
mfk_sheltering_nations_update = {
	if = {
		limit = {
			has_variable = mfk_nations_setup
			NOT = { has_variable = mfk_sheltering_nations_v2 }
		}
		set_variable = mfk_sheltering_nations_v2
		every_in_list = {
			variable = mfk_unaligned_list
			remove_character_role = character_role_nation_chief
			retire_character = yes
		}
		clear_variable_list = mfk_unaligned_list
		mfk_create_nation_chief = { NATION = ohionon IG = ig_armed_forces CULTURE = algonquian RELIGION = animist LIST = mfk_unaligned_list IDEOLOGY = traditionalist }
		mfk_create_nation_chief = { NATION = erie_freemen IG = ig_trade_unions CULTURE = afro_american RELIGION = protestant LIST = mfk_unaligned_list IDEOLOGY = abolitionist }
		mfk_create_nation_chief = { NATION = niagara_metis IG = ig_petty_bourgeoisie CULTURE = metis RELIGION = catholic LIST = mfk_unaligned_list IDEOLOGY = moderate }
		mfk_create_nation_chief = { NATION = wabash_shawnee IG = ig_armed_forces CULTURE = algonquian RELIGION = animist LIST = mfk_unaligned_list IDEOLOGY = jingoist_leader }
	}
	else_if = {
		limit = { NOT = { has_variable = mfk_sheltering_nations_v2 } }
		set_variable = mfk_sheltering_nations_v2
	}
}

# The loyalty of each bloc of chiefs, which drives the Movement for the Great Law and the Dehorners. Scope = Iroquois.
mfk_council_weights_update = {
	save_scope_as = council_country
	# Loyalty of each bloc as a whole (0 to 1), ignoring votes. Drives the Movement for the Great Law and the Dehorners.
	set_variable = { name = mfk_gl_loyalty_score value = 0 }
	set_variable = { name = mfk_gl_loyalty_count value = 0 }
	every_in_list = {
		variable = mfk_great_law_list
		scope:council_country = { change_variable = { name = mfk_gl_loyalty_count add = 1 } }
		if = {
			limit = { loyalty >= 75 }
			scope:council_country = { change_variable = { name = mfk_gl_loyalty_score add = 2 } }
		}
		else_if = {
			limit = { loyalty >= 25 }
			scope:council_country = { change_variable = { name = mfk_gl_loyalty_score add = 1 } }
		}
	}
	set_variable = { name = mfk_gl_loyalty_ratio value = 0.5 }
	if = {
		limit = { var:mfk_gl_loyalty_count > 0 }
		set_variable = { name = mfk_gl_loyalty_ratio value = var:mfk_gl_loyalty_score }
		change_variable = { name = mfk_gl_loyalty_ratio divide = var:mfk_gl_loyalty_count }
		change_variable = { name = mfk_gl_loyalty_ratio divide = 2 }
	}
	set_variable = { name = mfk_sh_loyalty_score value = 0 }
	set_variable = { name = mfk_sh_loyalty_count value = 0 }
	every_in_list = {
		variable = mfk_unaligned_list
		scope:council_country = { change_variable = { name = mfk_sh_loyalty_count add = 1 } }
		if = {
			limit = { loyalty >= 75 }
			scope:council_country = { change_variable = { name = mfk_sh_loyalty_score add = 2 } }
		}
		else_if = {
			limit = { loyalty >= 25 }
			scope:council_country = { change_variable = { name = mfk_sh_loyalty_score add = 1 } }
		}
	}
	set_variable = { name = mfk_sh_loyalty_ratio value = 0.5 }
	if = {
		limit = { var:mfk_sh_loyalty_count > 0 }
		set_variable = { name = mfk_sh_loyalty_ratio value = var:mfk_sh_loyalty_score }
		change_variable = { name = mfk_sh_loyalty_ratio divide = var:mfk_sh_loyalty_count }
		change_variable = { name = mfk_sh_loyalty_ratio divide = 2 }
	}
}


# A disloyal Sheltering Nation may leave the Confederacy. Scope = Iroquois.
mfk_sheltered_departure_check = {
	if = {
		limit = {
			NOT = { has_variable = mfk_departure_cooldown }
			any_in_list = {
				variable = mfk_unaligned_list
				mfk_sheltered_chief_can_depart = yes
			}
		}
		random = {
			chance = 4
			random_in_list = {
				variable = mfk_unaligned_list
				limit = { mfk_sheltered_chief_can_depart = yes }
				save_scope_as = departing_chief
			}
			set_variable = {
				name = mfk_departure_cooldown
				years = 2
			}
			trigger_event = { id = mfk_tadodaho.6 popup = yes }
		}
	}
}

# The departing nation leaves; it may be sheltered again later. Scope = Iroquois.
mfk_sheltered_depart = {
	remove_list_variable = { name = mfk_unaligned_list target = scope:departing_chief }
	if = {
		limit = { scope:departing_chief.var:mfk_nation ?= flag:lenape }
		remove_variable = mfk_sheltered_lenape
	}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:wyandot }
			remove_variable = mfk_sheltered_wyandot
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:mississauga }
			remove_variable = mfk_sheltered_mississauga
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:stockbridge }
			remove_variable = mfk_sheltered_stockbridge
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:potawatomi }
			remove_variable = mfk_sheltered_potawatomi
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:nanticoke }
			remove_variable = mfk_sheltered_nanticoke
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:tutelo }
			remove_variable = mfk_sheltered_tutelo
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:brothertown }
			remove_variable = mfk_sheltered_brothertown
		}
		else_if = {
			limit = { scope:departing_chief.var:mfk_nation ?= flag:odawa }
			remove_variable = mfk_sheltered_odawa
		}
	scope:departing_chief = {
		remove_character_role = character_role_nation_chief
		retire_character = yes
	}
	mfk_change_council_harmony = { VALUE = -5 }
	mfk_council_weights_update = yes
}


# Grants land to the chief of the nation sheltered most recently. Scope = Iroquois
mfk_grant_lands_to_last_sheltered = {
	every_in_list = {
		variable = mfk_unaligned_list
		limit = {
			OR = {
				AND = { var:mfk_nation ?= flag:lenape root.var:mfk_last_sheltered_nation ?= flag:lenape }
				AND = { var:mfk_nation ?= flag:wyandot root.var:mfk_last_sheltered_nation ?= flag:wyandot }
				AND = { var:mfk_nation ?= flag:mississauga root.var:mfk_last_sheltered_nation ?= flag:mississauga }
				AND = { var:mfk_nation ?= flag:stockbridge root.var:mfk_last_sheltered_nation ?= flag:stockbridge }
				AND = { var:mfk_nation ?= flag:potawatomi root.var:mfk_last_sheltered_nation ?= flag:potawatomi }
				AND = { var:mfk_nation ?= flag:odawa root.var:mfk_last_sheltered_nation ?= flag:odawa }
				AND = { var:mfk_nation ?= flag:nanticoke root.var:mfk_last_sheltered_nation ?= flag:nanticoke }
				AND = { var:mfk_nation ?= flag:tutelo root.var:mfk_last_sheltered_nation ?= flag:tutelo }
				AND = { var:mfk_nation ?= flag:brothertown root.var:mfk_last_sheltered_nation ?= flag:brothertown }
			}
		}
		add_modifier = {
			name = mfk_granted_lands
			years = 10
		}
	}
}

# A newly sheltered nation brings its people. They settle in the Confederacy's newer lands if it has any, otherwise in its capital. Scope = Iroquois
mfk_shelter_settle_pops = {
	if = {
		limit = {
			any_scope_state = {
				NOT = { state_region = s:STATE_SIX_NATIONS }
			}
		}
		random_scope_state = {
			limit = {
				NOT = { state_region = s:STATE_SIX_NATIONS }
			}
			create_pop = {
				culture = $CULTURE$
				religion = $RELIGION$
				size = 10000
			}
		}
	}
	else = {
		capital ?= {
			create_pop = {
				culture = $CULTURE$
				religion = $RELIGION$
				size = 10000
			}
		}
	}
}

# Every two to four years (the first after one year) a nation fleeing the Union asks to settle among the Haudenosaunee. Scope = Iroquois
mfk_shelter_request_check = {
	if = {
		limit = { NOT = { has_variable = mfk_shelter_requests_started } }
		set_variable = mfk_shelter_requests_started
		set_variable = {
			name = mfk_shelter_request_cooldown
			years = 1
		}
	}
	if = {
		limit = {
			NOT = { has_variable = mfk_shelter_request_cooldown }
			OR = {
				AND = { NOT = { has_variable = mfk_sheltered_mississauga } NOT = { has_variable = mfk_refused_mississauga } }
				AND = { NOT = { has_variable = mfk_sheltered_odawa } NOT = { has_variable = mfk_refused_odawa } }
				AND = { NOT = { has_variable = mfk_sheltered_nanticoke } NOT = { has_variable = mfk_refused_nanticoke } }
				AND = { NOT = { has_variable = mfk_sheltered_tutelo } NOT = { has_variable = mfk_refused_tutelo } }
				AND = { NOT = { has_variable = mfk_sheltered_lenape } NOT = { has_variable = mfk_refused_lenape } }
				AND = { NOT = { has_variable = mfk_sheltered_stockbridge } NOT = { has_variable = mfk_refused_stockbridge } }
				AND = { NOT = { has_variable = mfk_sheltered_brothertown } NOT = { has_variable = mfk_refused_brothertown } }
				AND = { NOT = { has_variable = mfk_sheltered_potawatomi } NOT = { has_variable = mfk_refused_potawatomi } }
				AND = { NOT = { has_variable = mfk_sheltered_wyandot } NOT = { has_variable = mfk_refused_wyandot } }
			}
		}
		if = {
			limit = { always = yes }
			random_list = {
				10 = {
					trigger = { NOT = { has_variable = mfk_sheltered_mississauga } NOT = { has_variable = mfk_refused_mississauga } }
					set_variable = { name = mfk_last_sheltered_nation value = flag:mississauga }
				}
				10 = {
					trigger = { NOT = { has_variable = mfk_sheltered_odawa } NOT = { has_variable = mfk_refused_odawa } }
					set_variable = { name = mfk_last_sheltered_nation value = flag:odawa }
				}

				}
			}

			random_list = {
				1 = { set_variable = { name = mfk_shelter_request_cooldown years = 2 } }
