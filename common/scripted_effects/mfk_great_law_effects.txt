# The nations of the Great Law of Peace, and the unaligned nations who have settled among them.
# Scope = Iroquois.
mfk_create_nation_chief = {
	create_character = {
		female = no
		culture = cu:$CULTURE$
		religion = rel:$RELIGION$
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
		on_created = {
			set_variable = { name = mfk_nation value = flag:$NATION$ }
		}
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}

mfk_replace_nation_chief = {
	save_scope_as = dead_nation_chief
	owner ?= {
		if = {
			limit = { is_target_in_variable_list = { name = mfk_great_law_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_great_law_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_great_law_list }
		}
		else_if = {
			limit = { is_target_in_variable_list = { name = mfk_unaligned_list target = scope:dead_nation_chief } }
			remove_list_variable = { name = mfk_unaligned_list target = scope:dead_nation_chief }
			mfk_nation_successor = { LIST = mfk_unaligned_list }
		}
	}
}

mfk_nation_successor = {
	mfk_roll_chief_$LIST$ = yes
}

# Creates the successor with a given interest group and ideology. Scope = Iroquois.
mfk_nation_successor_create = {
	create_character = {
		female = no
		culture = scope:dead_nation_chief.culture
		religion = scope:dead_nation_chief.religion
		interest_group = $IG$
		ideology = ideology_$IDEOLOGY$
		home_region = STATE_SIX_NATIONS
		ig_leader = no
		save_scope_as = new_nation_chief
	}
	scope:new_nation_chief ?= {
		add_character_role = character_role_nation_chief
		set_variable = { name = mfk_nation value = scope:dead_nation_chief.var:mfk_nation }
	}
	add_to_variable_list = {
		name = $LIST$
		target = scope:new_nation_chief
	}
}
