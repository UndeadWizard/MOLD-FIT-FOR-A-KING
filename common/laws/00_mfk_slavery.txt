law_chimeric_slavery = {
	group = lawgroup_slavery
	parent = law_legacy_slavery # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/legacy_slavery.dds"

	progressiveness = -50

	# The NAU only sees it once the Verdigri Compromise is before the Commons or has become law
	is_visible = {
		OR = {
			NOT = { has_law = law_type:law_chimeric_question }
			has_variable = mfk_cq_bill_compromise
		}
	}

	disallowing_laws = {
		law_multicultural
		law_affirmative_action
	}

	# Refinement is impossible without plasm
	can_enact = {
		custom_tooltip = {
			text = mfk_cq_law_locked_tt
			OR = {
				NOT = { has_journal_entry = je_mfk_chimeric_question }
				has_variable = mfk_cq_bill_compromise
			}
		}
		custom_tooltip = {
			text = law_chimeric_slavery_plasm_tt
			market_capital ?= {
				market = {
					mg:plasm = {
						market_goods_sell_orders > 0
					}
				}
			}
		}
	}

	can_impose = {
		can_impose_law_default = yes
	}

	on_activate = {
		custom_tooltip = law_chimeric_slavery_refinement_tt
		custom_tooltip = law_chimeric_slavery_penal_tt
		custom_tooltip = law_chimeric_slavery_bound_tt

		if = {
			limit = { has_variable = slavery_recently_abolished }
			remove_variable = slavery_recently_abolished
			every_scope_state = {
				limit = { has_variable = former_slave_state }
				remove_variable = former_slave_state
			}
		}

		ig:ig_landowners ?= {
			switch_from_colonial_slavery_to_pro_slavery = yes
		}

		hidden_effect = {
			if = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_pro_slavery
						has_core_ideology = ideology:ideology_pro_slavery_colonial_movement
					}
				}
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_pro_slavery
						has_core_ideology = ideology:ideology_pro_slavery_colonial_movement
					}
					set_core_ideology = ideology_pro_slavery_movement
				}
			}
		}
	}

	modifier = {
		interest_group_ig_landowners_pol_str_mult = 0.25
		country_influence_mult = -0.10
	}

	# Only countries already holding Verdigri pursue it
	ai_will_do = {
		any_scope_pop = {
			culture = cu:verdigri
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

# The NAU's 1836 settlement. The Crown's 1833 Act is enforced for people, but Jackson has never ruled on
# whether a Verdigri is a person. In the Slave States (states that already hold slaves) Verdigri bondage
# goes on unregulated, and poor free Verdigri are bound into it. It cannot spread to new states.
law_chimeric_question = {
	group = lawgroup_slavery
	parent = law_legacy_slavery # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/legacy_slavery.dds"

	progressiveness = -30

	is_visible = {
		c:USA ?= ROOT
	}

	can_enact = {
		c:USA ?= ROOT
	}

	disallowing_laws = {
		law_multicultural
		law_affirmative_action
	}

	on_activate = {
		custom_tooltip = law_chimeric_question_slave_states_tt
		custom_tooltip = law_chimeric_question_binding_tt
		custom_tooltip = law_chimeric_question_vats_tt
	}

	ai_will_do = {
		always = no
	}
}
