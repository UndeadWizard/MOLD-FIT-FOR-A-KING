		}
		add = 1
	}

		limit = {
			has_law_or_variant = law_type:law_parliamentary_republic
		}
		add = 1
	}
	if = { # Tenpo - Hardliner: PB Marginalized
		limit = {
            ig:ig_petty_bourgeoisie = {
                is_marginal = yes
            }
		}
		add = 1
	}

		limit = {
            country_has_building_type_levels = {
                target = bt:building_wheat_farm
                value >= 40
            }
		}
		add = 1
	}
}

traditionalist_reformer_goals_completed = {
	value = 0
	if = { # Tenpo - Reformer: Urbanization
		limit = {
            any_scope_state = {
                state_region = {
                    is_homeland = cu:iroquoian
                }
                any_scope_building = {
                    is_building_type = building_urban_center
                    level >= 3
                }
                count >= 1
            }
		}
		add = 1
	}
	if = { # Tenpo - Reformer: Edo System: Lax
		limit = {
			has_law_or_variant = law_type:law_monarchy
		}
		add = 1
	}
	if = { # Tenpo - Reformer: PB Clout
		limit = {
            ig:ig_petty_bourgeoisie = {
                ig_clout >= 0.1
            }
		}
		add = 1
	}
	if = { # Tenpo - Reformer: Bureaucrats
		limit = {
            NOT = {
                has_law_or_variant = law_type:law_hereditary_bureaucrats
            }
		}
		add = 1
	}
}

brant_total_agenda_goals_completed = {
	value = 0
	# Clan Mother Agenda
	if = { # Clan Mothers: Council Harmony
		limit = {
			je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 70 }
		}
		add = 1
	}
	if = { # Clan Mothers: the Clan Mothers keep their voice
		limit = {
			has_law = law_type:law_clan_mother
		}
		add = 1
	}
	if = { # Clan Mothers: no Six Nations land lost
		limit = {
			owns_entire_state_region = STATE_SIX_NATIONS
		}
		add = 1
	}
	if = { # Clan Mothers: the Three Sisters
		limit = {
			country_has_building_type_levels = { target = bt:building_maize_farm value >= 20 }
		}
		add = 1
	}
	if = { # Clan Mothers: the merchants kept in check
		limit = {
			ig:ig_petty_bourgeoisie ?= { is_marginal = yes }
		}
		add = 1
	}
	# Grand Hoyaneh Agenda
	if = { # Grand Hoyaneh: Free Hand
		limit = {
			je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" <= 30 }
		}
		add = 1
	}
	if = { # Grand Hoyaneh: Hoyaneh Ascendancy
		limit = {
			has_law = law_type:law_hoyaneh_ascendancy
		}
		add = 1
	}
	if = { # Grand Hoyaneh: a standing force of warriors
		limit = {
			has_law_or_variant = law_type:law_professional_army
		}
		add = 1
	}
	if = { # Grand Hoyaneh: towns to rival the British
		limit = {
			any_scope_state = { state_region = { is_homeland = cu:iroquoian } any_scope_building = { is_building_type = building_urban_center level >= 3 } }
		}
		add = 1
	}
	if = { # Grand Hoyaneh: the merchants prosper
		limit = {
			ig:ig_petty_bourgeoisie ?= { ig_clout >= 0.1 }
		}
		add = 1
	}
	# Shared Agenda
	if = { # Shared: no famine
		limit = {
			NOT = { any_scope_state = { state_region = { is_homeland = cu:iroquoian } has_famine = yes } }
		}
		add = 1
	}
	if = { # Shared: the Covenant with the Crown
		limit = {
			c:GBR ?= { relations:root >= relations_threshold:cordial }
		}
		add = 1
	}
	if = { # Shared: able to defend the League
		limit = {
			army_power_projection > 100
		}
		add = 1
	}
}
