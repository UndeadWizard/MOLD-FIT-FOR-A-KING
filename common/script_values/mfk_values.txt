clanmother_hardliner_goals_completed = {
	value = 0
	if = { # Tenpo - Hardliner: Social Control
		limit = {
			has_law_or_variant = law_type:law_clan_mother
		}
		add = 1
	}
	if = { # Tenpo - Hardliner: Edo System: Strict
		limit = {
			has_law_or_variant = law_type:law_parliamentary_republic
		}
		add = 1
	}
	if = { # Tenpo - Hardliner: PB Marginalized
		limit = {
            ig:ig_petty_bourgeoisie = {
                is_marginal = yes
            }
		}
		add = 1
	}
	if = { # Tenpo - Hardliner: Rural
		limit = {
            country_has_building_type_levels = {
                target = bt:building_wheat_farm
                value >= 40
            }
		}
		add = 1
	}
}

traditionalist_reformer_goals_completed = {
	value = 0
	if = { # Tenpo - Reformer: Urbanization
		limit = {
            any_scope_state = {
                state_region = {
                    is_homeland = cu:iroquoian
                }
                any_scope_building = {
                    is_building_type = building_urban_center
                    level >= 3
                }
                count >= 1
            }
		}
		add = 1
	}
	if = { # Tenpo - Reformer: Edo System: Lax
		limit = {
			has_law_or_variant = law_type:law_monarchy
		}
		add = 1
	}
	if = { # Tenpo - Reformer: PB Clout
		limit = {
            ig:ig_petty_bourgeoisie = {
                ig_clout >= 0.1
            }
		}
		add = 1
	}
	if = { # Tenpo - Reformer: Bureaucrats
		limit = {
            NOT = {
                has_law_or_variant = law_type:law_hereditary_bureaucrats
            }
		}
		add = 1
	}
}

brant_total_agenda_goals_completed = {
	value = 0
	# Clan Mother Agenda
	if = { # Clan Mothers: Council Harmony
		limit = {
			je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 70 }
		}
		add = 1
	}
	if = { # Clan Mothers: the Clan Mothers keep their voice
		limit = {
			has_law = law_type:law_clan_mother
		}
		add = 1
	}
	if = { # Clan Mothers: no Six Nations land lost
		limit = {
			owns_entire_state_region = STATE_SIX_NATIONS
		}
		add = 1
	}
	if = { # Clan Mothers: the Three Sisters
		limit = {
			country_has_building_type_levels = { target = bt:building_maize_farm value >= 20 }
		}
		add = 1
	}
	if = { # Clan Mothers: the merchants kept in check
		limit = {
			ig:ig_petty_bourgeoisie ?= { is_marginal = yes }
		}
		add = 1
	}
	# Brant's Vision
	if = { # Brant's Vision: a plasm industry to rival the Union's
		limit = {
			country_has_building_type_levels = { target = bt:building_plasm_industry value >= 5 }
		}
		add = 1
	}
	if = { # Brant's Vision: railways to bind the Confederacy together
		limit = {
			country_has_building_type_levels = { target = bt:building_railway value >= 5 }
		}
		add = 1
	}
	if = { # Brant's Vision: a standing force of warriors
		limit = {
			has_law_or_variant = law_type:law_professional_army
		}
		add = 1
	}
	if = { # Brant's Vision: towns to rival the British
		limit = {
			any_scope_state = { state_region = { is_homeland = cu:iroquoian } any_scope_building = { is_building_type = building_urban_center level >= 3 } }
		}
		add = 1
	}
	if = { # Brant's Vision: the merchants prosper
		limit = {
			ig:ig_petty_bourgeoisie ?= { ig_clout >= 0.1 }
		}
		add = 1
	}
	# Shared Agenda
	if = { # Shared: no famine
		limit = {
			NOT = { any_scope_state = { state_region = { is_homeland = cu:iroquoian } has_famine = yes } }
		}
		add = 1
	}
	if = { # Shared: the Covenant with the Crown
		limit = {
			c:GBR ?= { relations:root >= relations_threshold:cordial }
		}
		add = 1
	}
	if = { # Shared: a growing economy
		limit = {
			has_variable = mfk_brant_gdp_goal
			gdp >= var:mfk_brant_gdp_goal
		}
		add = 1
	}
	if = { # Shared: able to defend the League
		limit = {
			army_power_projection > 100
		}
		add = 1
	}
}

# Disloyal Great Law chiefs rally their nations to the Movement for the Great Law. 1.5 when all are disloyal, 0.5 when all are loyal. Scope = Iroquois
mfk_great_law_movement_factor = {
	value = 1
	if = {
		limit = { has_variable = mfk_gl_loyalty_ratio }
		value = 1.5
		subtract = var:mfk_gl_loyalty_ratio
	}
	min = 0.5
	max = 1.5
}

# Loyal Sheltering chiefs bring their nations behind the Brantist Movement. 0.5 when all are disloyal, 1.5 when all are loyal. Scope = Iroquois
mfk_dehorners_movement_factor = {
	value = 1
	if = {
		limit = { has_variable = mfk_sh_loyalty_ratio }
		value = 0.5
		add = var:mfk_sh_loyalty_ratio
	}
	min = 0.5
	max = 1.5
}

mfk_great_law_movement_pct = {
	value = mfk_great_law_movement_factor
	subtract = 1
	multiply = 100
}

mfk_dehorners_movement_pct = {
	value = mfk_dehorners_movement_factor
	subtract = 1
	multiply = 100
}
