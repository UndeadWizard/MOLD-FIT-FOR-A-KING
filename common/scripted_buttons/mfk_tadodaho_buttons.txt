# Spend Council Harmony to push the law currently being enacted through the Council
mfk_press_the_council_button = {
	name = "mfk_press_the_council_button"
	desc = "mfk_press_the_council_button_desc"

	visible = {
		has_journal_entry = je_tadodaho
	}

	cooldown = { months = 6 }

	possible = {
		custom_tooltip = {
			text = mfk_press_council_enacting_tt
			exists = currently_enacting_law
		}
		custom_tooltip = {
			text = mfk_press_council_harmony_tt
			je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 20 }
		}
		custom_tooltip = {
			text = mfk_press_council_active_tt
			NOT = { has_modifier = mfk_council_pressed }
		}
	}

	effect = {
		mfk_change_council_harmony = { VALUE = -10 }
		add_modifier = {
			name = mfk_council_pressed
			months = 6
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				exists = currently_enacting_law
				je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 50 }
			}
			add = 20
		}
	}
}

# Council Fire: rebuild the council longhouse at Onondaga
mfk_reopen_longhouse_button = {
	name = "mfk_reopen_longhouse_button"
	desc = "mfk_reopen_longhouse_button_desc"
	icon = "gfx/interface/icons/ig_icons/landowners.dds"

	visible = {
		has_journal_entry = je_john_brant_rule
	}

	possible = {
		hidden_trigger = { NOT = { has_variable = mfk_reopen_longhouse_button_done } }
		custom_tooltip = {
			text = mfk_owns_six_nations_tt
			owns_entire_state_region = STATE_SIX_NATIONS
		}
		custom_tooltip = {
			text = mfk_harmony_40_tt
			je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 40 }
		}
		ig:ig_devout ?= { is_marginal = no }
	}

	effect = {
		custom_tooltip = {
			text = mfk_antlers_fire_medium_tt
			je:je_john_brant_rule ?= {
				add_progress = { value = -18 name = mfk_antlers_balance_bar }
			}
		}
		mfk_change_council_harmony = { VALUE = 10 }
		add_modifier = {
			name = mfk_onondaga_longhouse_reopened
			years = 20
		}
		set_variable = mfk_reopen_longhouse_button_done
	}

	selected = {
		has_variable = mfk_reopen_longhouse_button_done
	}

	ai_chance = {
		value = 30
	}
}
