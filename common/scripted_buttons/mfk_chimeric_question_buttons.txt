# With a British Governor-General, Philadelphia can ask the Crown to weigh in
mfk_cq_petition_whitehall_button = {
	name = "mfk_cq_petition_whitehall_button"
	desc = "mfk_cq_petition_whitehall_button_desc"

	visible = {
		NOT = { has_variable = mfk_gg_last_was_american }
	}

	possible = {
		custom_tooltip = {
			text = mfk_cq_question_compromise_lean_tt
			je:je_mfk_chimeric_question ?= {
				"scripted_bar_progress(mfk_cq_question_bar)" >= 45
			}
		}
	}

	cooldown = { days = 730 }

	ai_chance = {
		value = 10
	}

	effect = {
		je:je_mfk_chimeric_question ?= {
			add_progress = { value = -8 name = mfk_cq_question_bar }
			add_progress = { value = 10 name = mfk_cq_tension_bar }
		}
		change_variable = { name = mfk_cq_shift_var add = -0.4 }
		# A Union in debt to London listens harder to Whitehall
		if = {
			limit = { has_variable = mfk_boe_loan }
			je:je_mfk_chimeric_question ?= {
				add_progress = { value = -4 name = mfk_cq_question_bar }
			}
		}
		add_modifier = {
			name = mfk_cq_lobbying_costs
			years = 1
			multiplier = money_amount_multiplier_medium
		}
		c:GBR ?= {
			change_relations = {
				country = ROOT
				value = 10
			}
		}
	}
}

# Bring a bill to settle the Chimeric Question before the Commons. It is enacted like any law,
# until the decisive vote (mfk_cq.10) cuts the process short one way or the other.
mfk_cq_compromise_bill_button = {
	name = "mfk_cq_compromise_bill_button"
	desc = "mfk_cq_compromise_bill_button_desc"

	visible = {
		NOT = { has_variable = mfk_cq_bill }
	}

	possible = {
		custom_tooltip = {
			text = mfk_cq_question_compromise_60_tt
			je:je_mfk_chimeric_question ?= {
				"scripted_bar_progress(mfk_cq_question_bar)" >= 60
			}
		}
		ig:ig_landowners ?= {
			is_in_government = yes
		}
		custom_tooltip = {
			text = mfk_cq_bill_cooldown_tt
			NOT = { has_variable = mfk_cq_bill_cooldown }
		}
	}

	ai_chance = {
		value = 20
	}

	effect = {
		set_variable = mfk_cq_bill
		set_variable = mfk_cq_bill_compromise
		cancel_enactment = yes
		start_enactment = law_type:law_chimeric_slavery
		add_enactment_modifier = {
			name = mfk_cq_bill_backing
		}
	}
}
