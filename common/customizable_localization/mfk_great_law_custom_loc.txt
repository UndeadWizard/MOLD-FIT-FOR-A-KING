mfk_council_consensus_tier = {
	type = country
	text = {
		trigger = {
			has_variable = mfk_consensus_count
			var:mfk_consensus_count >= 5
		}
		localization_key = mfk_consensus_tier_unanimity
	}
	text = {
		trigger = {
			has_variable = mfk_consensus_count
			var:mfk_consensus_count >= 3
		}
		localization_key = mfk_consensus_tier_partial
	}
	text = {
		trigger = { always = yes }
		localization_key = mfk_consensus_tier_deadlock
	}
}
