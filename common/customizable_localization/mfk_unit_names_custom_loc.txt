	text = {
		trigger = { mfk_breeds_native = yes }
		localization_key = mfk_unit_war_hounds_bear_claw
	}
	text = {
		trigger = { mfk_breeds_louisiane = yes }
		localization_key = mfk_unit_war_hounds_bayou
	}
	text = {
		trigger = { mfk_breeds_cavalier = yes }

	text = {

}
