# Mold Fit for a King: the Tadodaho keeps the Onondaga council fire, independent of the Grand Hoyaneh
je_tadodaho = {
	group = je_group_great_law

	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	activation_description = {
		always = yes
	}

	widget = {
		gui = "gui/journal_entry_widgets/mfk_tadodaho_widgets.gui"
		name = "widget_je_tadodaho"
		container = "custom_widget_container_je_icon"
	}

	widget = {
		gui = "gui/journal_entry_widgets/mfk_great_law_widgets.gui"
		name = "widget_je_great_law_nations"
		container = "custom_widget_container_3"
	}

	scripted_progress_bar = mfk_council_harmony_bar

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { exists = c:IRO }
		}
	}

	weight = 100

	transferable = no

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_dehorning_crisis = {
	group = je_group_great_law

	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	activation_description = {
		always = yes
	}


	timeout = 730

	outcome = { # The council and the Grand Hoyaneh are reconciled
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = mfk_dehorning_reconciled_tt
				je:je_tadodaho ?= { "scripted_bar_progress(mfk_council_harmony_bar)" >= 50 }
			}
		}
		effect = {
			add_modifier = {
				name = mfk_council_reconciled
				years = 3
			}
			mfk_dehorning_cleanup = yes
		}
	}

	outcome = { # Time has run out: the antlers must be decided
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			trigger_event = { id = mfk_tadodaho.4 popup = yes }
		}
	}

	outcome = { # The Grand Hoyaneh is no longer the same man, or the Confederacy is gone
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { exists = c:IRO }
				NOT = { exists = var:mfk_dehorning_target }
				NOT = { ruler ?= var:mfk_dehorning_target }
			}
		}
		effect = {
			mfk_dehorning_cleanup = yes
		}
	}

	weight = 200

	transferable = no

	should_be_pinned_by_default_uninvolved_or_context = yes
}
