# The dehorning crisis comes to a head
mfk_tadodaho.4 = {
	type = country_event
	placement = root

	title = mfk_tadodaho.4.t
	desc = mfk_tadodaho.4.d
	flavor = mfk_tadodaho.4.f

	event_image = {
		video = "europenorthamerica_native_american"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	duration = 3

	trigger = {
		exists = var:mfk_dehorning_target
	}

	immediate = {
		random_scope_character = {
			limit = { has_role = character_role_tadodaho }
			save_scope_as = tadodaho_scope
		}
		var:mfk_dehorning_target ?= { save_scope_as = hoyaneh_scope }
	}

	option = { # The antlers are taken
		name = mfk_tadodaho.4.a
		default_option = yes
		custom_tooltip = mfk_dehorning_removed_tt
		scope:hoyaneh_scope ?= {
			retire_character = yes
		}
		mfk_change_council_harmony = { VALUE = 30 }
		mfk_dehorning_cleanup = yes
		ai_chance = { base = 25 }
	}

	option = { # The Grand Hoyaneh submits to the council
		name = mfk_tadodaho.4.b
		mfk_change_council_harmony = { VALUE = 25 }
		scope:hoyaneh_scope ?= {
			interest_group ?= {
				add_modifier = {
					name = mfk_hoyaneh_humbled
					years = 3
				}
			}
		}
		mfk_dehorning_cleanup = yes
		ai_chance = { base = 50 }
	}

	option = { # The Grand Hoyaneh calls on the Crown
		name = mfk_tadodaho.4.c
		if = {
			limit = { exists = c:GBR }
			change_relations = {
				country = c:GBR
				value = 20
			}
		}
		add_modifier = {
			name = mfk_confederacy_divided
			years = 5
		}
		mfk_change_council_harmony = { VALUE = -10 }
		mfk_dehorning_cleanup = yes
		ai_chance = { base = 25 }
	}
}

# John Brant fails to reconcile the clans and loses the Antlers of Office; whichever side holds the balance takes them
mfk_tadodaho.7 = {
	type = country_event
	placement = root

	title = mfk_tadodaho.7.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:mfk_antlers_winner ?= flag:dehorners }
				desc = mfk_tadodaho.7.d_dehorners
			}
			triggered_desc = {
				desc = mfk_tadodaho.7.d_fire
			}
		}
	}
	flavor = mfk_tadodaho.7.f

	event_image = {
		video = "europenorthamerica_native_american"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	duration = 3

	immediate = {
		ruler ?= { save_scope_as = fallen_hoyaneh }
		random_scope_character = {
			limit = { has_role = character_role_tadodaho }
			save_scope_as = tadodaho_scope
		}
		if = {
			limit = {
				NOT = { any_scope_character = { has_template = IRO_alexander_norton } }
			}
			create_character = {
				template = IRO_alexander_norton
				save_scope_as = norton_scope
			}
		}
		else = {
			random_scope_character = {
				limit = { has_template = IRO_alexander_norton }
				save_scope_as = norton_scope
			}
		}
	}

	# The Council Fire held the balance: the Tadodaho takes up the Antlers
	option = {
		name = mfk_tadodaho.7.a
		default_option = yes
		trigger = {
			NOT = { var:mfk_antlers_winner ?= flag:dehorners }
			exists = scope:tadodaho_scope
		}
		set_variable = mfk_tadodaho_rules
		set_state_religion = rel:animist
		scope:tadodaho_scope = {
			set_character_as_ruler = yes
		}
		add_modifier = {
			name = mfk_fire_holds_the_antlers
			years = 10
		}
		mfk_change_council_harmony = { VALUE = 20 }
		scope:fallen_hoyaneh ?= {
			retire_character = yes
		}
	}

	# The Dehorners held the balance: Alexander Norton seizes the Antlers
	option = {
		name = mfk_tadodaho.7.b
		default_option = yes
		trigger = {
			OR = {
				var:mfk_antlers_winner ?= flag:dehorners
				NOT = { exists = scope:tadodaho_scope }
			}
			exists = scope:norton_scope
		}
		scope:norton_scope = {
			set_character_as_ruler = yes
		}
		add_modifier = {
			name = mfk_nortons_war_council
			years = 10
		}
		mfk_change_council_harmony = { VALUE = -20 }
		scope:fallen_hoyaneh ?= {
			retire_character = yes
		}
	}
}

# A nation fleeing the Union asks to settle among the Haudenosaunee
mfk_tadodaho.8 = {
	type = country_event
	placement = root

	title = mfk_tadodaho.8.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { mfk_requesting_nation_is_traditional = yes }
				desc = mfk_tadodaho.8.d_traditional
			}
			triggered_desc = {
				desc = mfk_tadodaho.8.d_christian
			}
		}
	}
	flavor = mfk_tadodaho.8.f

	event_image = {
		video = "europenorthamerica_native_american"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	duration = 3

	trigger = {
		has_variable = mfk_last_sheltered_nation
	}

	# Take them in
	option = {
		name = mfk_tadodaho.8.a
		default_option = yes
		mfk_shelter_requesting_nation = yes
		if = {
			limit = { mfk_requesting_nation_is_traditional = yes }
			ig:ig_devout ?= {
				add_modifier = {
					name = mfk_clan_mothers_kin_welcomed
					years = 5
				}
			}
			mfk_change_council_harmony = { VALUE = 5 }
			je:je_john_brant_rule ?= {
				add_progress = { value = -12 name = mfk_antlers_balance_bar }
			}
		}
		else = {
			ig:ig_devout ?= {
				add_modifier = {
					name = mfk_clan_mothers_strangers_welcomed
					years = 5
				}
			}
			mfk_change_council_harmony = { VALUE = -10 }
			je:je_john_brant_rule ?= {
				add_progress = { value = 12 name = mfk_antlers_balance_bar }
			}
		}
		hidden_effect = { mfk_council_weights_update = yes }
		ai_chance = {
			base = 50
			modifier = {
				trigger = { mfk_requesting_nation_is_traditional = yes }
				add = 50
			}
		}
	}

	# Take them in on the Grand Hoyaneh's own lands, binding them to him
	option = {
		name = mfk_tadodaho.8.b
		mfk_shelter_requesting_nation = yes
		mfk_grant_lands_to_last_sheltered = yes
		every_in_list = {
			variable = mfk_great_law_list
			add_modifier = {
				name = mfk_slighted_by_grant
				years = 5
			}
		}
		if = {
			limit = { mfk_requesting_nation_is_traditional = no }
			ig:ig_devout ?= {
				add_modifier = {
					name = mfk_clan_mothers_strangers_welcomed
					years = 5
				}
			}
		}
		mfk_change_council_harmony = { VALUE = -15 }
		je:je_john_brant_rule ?= {
			add_progress = { value = 24 name = mfk_antlers_balance_bar }
		}
		hidden_effect = { mfk_council_weights_update = yes }
		ai_chance = {
			base = 15
			modifier = {
				trigger = { has_law = law_type:law_hoyaneh_ascendancy }
				add = 35
			}
		}
	}

	# Turn them away
	option = {
		name = mfk_tadodaho.8.c
		hidden_effect = {
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:mississauga }
				set_variable = { name = mfk_refused_mississauga years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:odawa }
				set_variable = { name = mfk_refused_odawa years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:nanticoke }
				set_variable = { name = mfk_refused_nanticoke years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:tutelo }
				set_variable = { name = mfk_refused_tutelo years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:lenape }
				set_variable = { name = mfk_refused_lenape years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:stockbridge }
				set_variable = { name = mfk_refused_stockbridge years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:brothertown }
				set_variable = { name = mfk_refused_brothertown years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:potawatomi }
				set_variable = { name = mfk_refused_potawatomi years = 10 }
			}
			if = {
				limit = { var:mfk_last_sheltered_nation ?= flag:wyandot }
				set_variable = { name = mfk_refused_wyandot years = 10 }
			}
		}
		if = {
			limit = { mfk_requesting_nation_is_traditional = yes }
			ig:ig_devout ?= {
				add_modifier = {
					name = mfk_clan_mothers_kin_refused
					years = 3
				}
			}
			mfk_change_council_harmony = { VALUE = -5 }
			je:je_john_brant_rule ?= {
				add_progress = { value = 6 name = mfk_antlers_balance_bar }
			}
		}
		else = {
			mfk_change_council_harmony = { VALUE = 3 }
			je:je_john_brant_rule ?= {
				add_progress = { value = -6 name = mfk_antlers_balance_bar }
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					mfk_requesting_nation_is_traditional = no
					has_law = law_type:law_clan_mother
				}
				add = 40
			}
		}
	}
}

# John Brant completes his agenda: the reward follows the path he leaned toward
mfk_tadodaho.11 = {
	type = country_event
	placement = root

	title = mfk_tadodaho.11.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { mfk_brant_clan_mother_goals > mfk_brant_grand_hoyaneh_goals }
				desc = mfk_tadodaho.11.d_clan_mother
			}
			triggered_desc = {
				trigger = { mfk_brant_grand_hoyaneh_goals > mfk_brant_clan_mother_goals }
				desc = mfk_tadodaho.11.d_grand_hoyaneh
			}
			triggered_desc = {
				desc = mfk_tadodaho.11.d_covenant
			}
		}
	}
	flavor = mfk_tadodaho.11.f

	event_image = {
		video = "europenorthamerica_native_american"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/faction_icons/faction_grand_hoyaneh.dds"

	duration = 3

	# More Clan Mother goals: the Great Law restored
	option = {
		name = mfk_tadodaho.11.a
		default_option = yes
		trigger = { mfk_brant_clan_mother_goals > mfk_brant_grand_hoyaneh_goals }
		if = {
			limit = { has_modifier = mfk_brants_covenant }
			remove_modifier = mfk_brants_covenant
		}
		add_modifier = { name = mfk_great_law_restored }
		mfk_change_council_harmony = { VALUE = 20 }
	}

	# More Grand Hoyaneh goals: Brant's Ascendancy
	option = {
		name = mfk_tadodaho.11.b
		default_option = yes
		trigger = { mfk_brant_grand_hoyaneh_goals > mfk_brant_clan_mother_goals }
		if = {
			limit = { has_modifier = mfk_brants_covenant }
			remove_modifier = mfk_brants_covenant
		}
		add_modifier = { name = mfk_brants_ascendancy }
	}

	# Evenly matched: the Covenant endures
	option = {
		name = mfk_tadodaho.11.c
		default_option = yes
		trigger = {
			NOT = { mfk_brant_clan_mother_goals > mfk_brant_grand_hoyaneh_goals }
			NOT = { mfk_brant_grand_hoyaneh_goals > mfk_brant_clan_mother_goals }
		}
	}
}
