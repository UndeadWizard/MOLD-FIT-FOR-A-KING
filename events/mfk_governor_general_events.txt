# The NAU learns who its new Governor-General is
mfk_gg.2 = {
	type = country_event
	placement = root

	title = mfk_gg.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:new_crown_favourite ?= { has_variable = mfk_british_peer } }
				desc = mfk_gg.2.d_peer
			}
			triggered_desc = {
				desc = mfk_gg.2.d_american
			}
		}
	}
	flavor = mfk_gg.2.f

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = scope:new_crown_favourite
	}

	option = { # Accept the Crown's choice
		name = mfk_gg.2.a
		default_option = yes
	}

	option = { # Protest to Whitehall
		name = mfk_gg.2.b
		change_relations = {
			country = c:GBR
			value = -15
		}
		ig:ig_rural_folk ?= {
			if = {
				limit = { has_modifier = crown_favourite_resentment }
				remove_modifier = crown_favourite_resentment
			}
		}
		ai_chance = { base = 1 }
	}
}

# A new President-General takes office and Whitehall sends its confirmation (Great Britain)
mfk_gg.3 = {
	type = country_event
	placement = root

	title = mfk_gg.3.t
	desc = mfk_gg.3.d
	flavor = mfk_gg.3.f

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = c:USA
		exists = scope:new_president_general
		scope:new_president_general = { is_ruler_of_own_country = yes } # still in office
	}

	option = { # Royal congratulations
		name = mfk_gg.3.a
		trigger = { mfk_gbr_trusts_nau = yes }
		change_relations = {
			country = c:USA
			value = 10
		}
		scope:new_president_general = {
			add_modifier = {
				name = mfk_crown_blessing
				years = 2
			}
			interest_group ?= {
				add_modifier = {
					name = mfk_investiture_blessing_ig
					years = 2
				}
			}
		}
		hidden_effect = {
			c:USA = {
				set_variable = { name = mfk_gg_warm_confirmation days = 30 }
				trigger_event = { id = mfk_gg.5 popup = yes }
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { "relations(c:USA)" >= relations_threshold:amicable }
				add = 30
			}
		}
	}

	option = { # A formal confirmation
		name = mfk_gg.3.b
		default_option = yes
		custom_tooltip = mfk_gg_formal_confirmation_tt
		hidden_effect = {
			c:USA = { trigger_event = { id = mfk_gg.5 popup = yes } }
		}
		ai_chance = { base = 20 }
	}

	option = { # A cold confirmation
		name = mfk_gg.3.c
		trigger = {
			OR = {
				NOT = { mfk_gbr_trusts_nau = yes }
				scope:new_president_general.interest_group ?= {
					NOT = { mfk_ig_serves_crown = yes }
				}
			}
		}
		change_relations = {
			country = c:USA
			value = -10
		}
		scope:new_president_general = {
			add_modifier = {
				name = mfk_crown_snub
				years = 2
			}
		}
		custom_tooltip = mfk_gg_cold_favour_tt
		hidden_effect = {
			c:USA = {
				mfk_gg_favour_loyal_opposition = yes
				trigger_event = { id = mfk_gg.4 popup = yes }
			}
		}
		ai_chance = {
			base = 5
			modifier = {
				trigger = { "relations(c:USA)" <= relations_threshold:poor }
				add = 40
			}
		}
	}
}
