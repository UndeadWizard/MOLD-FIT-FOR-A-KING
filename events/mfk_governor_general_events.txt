# The NAU learns who its new Governor-General is
mfk_gg.2 = {
	type = country_event
	placement = root

	title = mfk_gg.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:new_crown_favourite ?= { has_variable = mfk_british_peer } }
				desc = mfk_gg.2.d_peer
			}
			triggered_desc = { # The first American Governor-General
				trigger = { NOT = { has_variable = mfk_gg_had_american } }
				desc = mfk_gg.2.d_american
			}
			triggered_desc = { # Another American straight after an American
				trigger = { has_variable = mfk_gg_last_was_american }
				desc = mfk_gg.2.d_american_again
			}
			triggered_desc = { # An American again, after a British peer
				desc = mfk_gg.2.d_american_return
			}
		}
	}
	flavor = mfk_gg.2.f

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = scope:new_crown_favourite
	}

	option = { # Accept the Crown's choice
		name = mfk_gg.2.a
		default_option = yes
		c:GBR ?= {
			change_relations = {
				country = ROOT
				value = 10
			}
		}
		scope:new_crown_favourite.interest_group ?= {
			add_modifier = {
				name = mfk_royal_patronage
				years = 2
			}
		}
		if = {
			limit = {
				ig:ig_rural_folk ?= { has_modifier = crown_favourite_resentment }
			}
			custom_tooltip = mfk_gg.2.a_resentment_tt
		}
	}

	option = { # Protest to Whitehall
		name = mfk_gg.2.b
		change_relations = {
			country = c:GBR
			value = -15
		}
		ig:ig_rural_folk ?= {
			if = {
				limit = { has_modifier = crown_favourite_resentment }
				remove_modifier = crown_favourite_resentment
			}
		}
		ai_chance = { base = 1 }
	}

	# Remember whether this Governor-General was American, for the next appointment's text
	after = {
		if = {
			limit = {
				scope:new_crown_favourite ?= { has_variable = mfk_british_peer }
			}
			remove_variable = mfk_gg_last_was_american
		}
		else = {
			set_variable = mfk_gg_last_was_american
			set_variable = mfk_gg_had_american
		}
	}
}

# Reserving a bill. Root = Great Britain. The Union has begun enacting a law the Crown objects to, and the
# Governor-General asks London whether to reserve it for His Majesty's pleasure.
mfk_gg.20 = {
	type = country_event
	placement = root

	title = mfk_gg.20.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:governor_general ?= { has_variable = mfk_british_peer } }
				desc = mfk_gg.20.d_peer
			}
			triggered_desc = {
				desc = mfk_gg.20.d_american
			}
		}
	}
	flavor = mfk_gg.20.f

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		c:USA ?= {
			mfk_crown_would_reserve_law = yes
			NOT = { has_variable = mfk_bill_reserved }
		}
	}

	immediate = {
		c:USA = {
			save_scope_as = nau_scope
			currently_enacting_law = { save_scope_as = reserved_law }
			random_scope_character = {
				limit = { has_role = character_role_governor_general }
				save_scope_as = governor_general
			}
		}
	}

	# Reserve the bill
	option = {
		name = mfk_gg.20.a
		set_variable = {
			name = mfk_reservation_cooldown
			years = 2
		}
		scope:nau_scope = {
			set_variable = mfk_bill_reserved
			add_modifier = { name = mfk_bill_reserved }
			custom_tooltip = mfk_gg_bill_reserved_tt
			hidden_effect = {
				trigger_event = { id = mfk_gg.21 popup = yes }
			}
		}
		change_relations = {
			country = scope:nau_scope
			value = -5
		}
		# An American Governor-General reserving against his own countrymen
		if = {
			limit = { scope:governor_general ?= { NOT = { has_variable = mfk_british_peer } } }
			scope:nau_scope = { add_liberty_desire = 5 }
			scope:governor_general.interest_group ?= {
				add_modifier = {
					name = mfk_gg_reserved_against_countrymen
					years = 2
				}
			}
		}
		hidden_effect = {
			trigger_event = { id = mfk_gg.22 days = 90 popup = yes }
		}
		ai_chance = {
			base = 30
			modifier = {
				trigger = { scope:governor_general ?= { has_variable = mfk_british_peer } }
				add = 30
			}
		}
	}

	# Let the Commons decide
	option = {
		name = mfk_gg.20.b
		default_option = yes
		ai_chance = {
			base = 40
			modifier = {
				trigger = { scope:governor_general ?= { NOT = { has_variable = mfk_british_peer } } }
				add = 30
			}
		}
	}
}
