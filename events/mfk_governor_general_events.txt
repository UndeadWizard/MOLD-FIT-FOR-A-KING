# The NAU learns who its new Governor-General is
mfk_gg.2 = {
	type = country_event
	placement = root

	title = mfk_gg.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:new_crown_favourite ?= { has_variable = mfk_british_peer } }
				desc = mfk_gg.2.d_peer
			}
			triggered_desc = {
				desc = mfk_gg.2.d_american
			}
		}
	}
	flavor = mfk_gg.2.f

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	trigger = {
		exists = scope:new_crown_favourite
	}

	option = { # Accept the Crown's choice
		name = mfk_gg.2.a
		default_option = yes
	}

	option = { # Protest to Whitehall
		name = mfk_gg.2.b
		change_relations = {
			country = c:GBR
			value = -15
		}
		ig:ig_rural_folk ?= {
			if = {
				limit = { has_modifier = crown_favourite_resentment }
				remove_modifier = crown_favourite_resentment
			}
		}
		ai_chance = { base = 1 }
	}
}
