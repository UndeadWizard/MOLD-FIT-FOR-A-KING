namespace = mfk_e1836

# The Election of 1836 is decided (NAU)
mfk_e1836.1 = {
	type = country_event
	placement = root

	title = mfk_e1836.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = mfk_e1836_winner_jacksonian
					scope:mfk_new_pg ?= { has_template = usa_andrew_jackson_template }
				}
				desc = mfk_e1836.1.d_jacksonian_jackson
			}
			triggered_desc = {
				trigger = { has_variable = mfk_e1836_winner_jacksonian }
				desc = mfk_e1836.1.d_jacksonian
			}
			triggered_desc = {
				trigger = { has_variable = mfk_e1836_winner_whig }
				desc = mfk_e1836.1.d_whig
			}
			triggered_desc = {
				trigger = { has_variable = mfk_e1836_winner_barnburner }
				desc = mfk_e1836.1.d_barnburner
			}
		}
	}
	flavor = mfk_e1836.1.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"

	duration = 3

	fire_only_once = yes

	# The winners form the government
	immediate = {
		add_modifier = {
			name = mfk_commons_has_spoken
			years = 2
		}
		# The result event announces the President-General itself
		set_variable = { name = hide_ruler_change_notification days = 3 }
		# The coalition gives way to the winners' mandate
		if = {
			limit = { has_modifier = mfk_jacksons_coalition }
			remove_modifier = mfk_jacksons_coalition
		}
		if = {
			limit = { has_variable = mfk_e1836_winner_jacksonian }
			set_ruling_interest_groups = { ig_rural_folk ig_landowners }
			add_modifier = {
				name = mfk_e1836_mandate_jacksonian
				years = 2
			}
			ig:ig_rural_folk.leader ?= {
				save_scope_as = mfk_new_pg
				set_character_as_ruler = yes
			}
		}
		else_if = {
			limit = { has_variable = mfk_e1836_winner_whig }
			set_ruling_interest_groups = { ig_industrialists ig_petty_bourgeoisie }
			add_modifier = {
				name = mfk_e1836_mandate_whig
				years = 2
			}
			ig:ig_industrialists.leader ?= {
				save_scope_as = mfk_new_pg
				set_character_as_ruler = yes
			}
		}
		else_if = {
			limit = { has_variable = mfk_e1836_winner_barnburner }
			set_ruling_interest_groups = { ig_intelligentsia ig_industrialists }
			add_modifier = {
				name = mfk_e1836_mandate_barnburner
				years = 2
			}
			ig:ig_intelligentsia.leader ?= {
				save_scope_as = mfk_new_pg
				set_character_as_ruler = yes
			}
		}
	}

	option = {
		name = mfk_e1836.1.jacksonian
		default_option = yes
		trigger = { has_variable = mfk_e1836_winner_jacksonian }
		custom_tooltip = mfk_e1836.1.jacksonian_tt
		if = {
			limit = { NOT = { has_variable = mfk_cq_shift_var } }
			set_variable = { name = mfk_cq_shift_var value = 0 }
		}
		change_variable = { name = mfk_cq_shift_var add = 0.5 }
	}

	option = {
		name = mfk_e1836.1.whig
		default_option = yes
		trigger = { has_variable = mfk_e1836_winner_whig }
		custom_tooltip = mfk_e1836.1.whig_tt
		if = {
			limit = { NOT = { has_variable = mfk_cq_shift_var } }
			set_variable = { name = mfk_cq_shift_var value = 0 }
		}
		change_variable = { name = mfk_cq_shift_var add = -0.25 }
	}

	option = {
		name = mfk_e1836.1.barnburner
		default_option = yes
		trigger = { has_variable = mfk_e1836_winner_barnburner }
		custom_tooltip = mfk_e1836.1.barnburner_tt
		if = {
			limit = { NOT = { has_variable = mfk_cq_shift_var } }
			set_variable = { name = mfk_cq_shift_var value = 0 }
		}
		change_variable = { name = mfk_cq_shift_var add = -0.75 }
	}
}
