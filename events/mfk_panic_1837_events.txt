# The banks suspend payment (NAU, 10 May 1837)
mfk_p1837.1 = {
	type = country_event
	placement = root

	title = mfk_p1837.1.t
	desc = mfk_p1837.1.d
	flavor = mfk_p1837.1.f

	event_image = {
		video = "unspecific_factory_closed"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	fire_only_once = yes

	immediate = {
		set_global_variable = mfk_panic_1837_var
		save_scope_as = nau_scope
	}

	option = {
		name = mfk_p1837.1.a
		default_option = yes
		if = {
			limit = {
				exists = c:GBR
				is_subject_of = c:GBR
			}
			custom_tooltip = mfk_p1837.1.a_tt
			c:GBR = {
				trigger_event = { id = mfk_p1837.4 popup = yes }
			}
		}
		else = {
			trigger_event = { id = mfk_p1837.5 days = 7 popup = yes }
		}
	}

}

# The Cotton Collapse (NAU, six months into the crash)
mfk_p1837.2 = {
	type = country_event
	placement = root

	title = mfk_p1837.2.t
	desc = mfk_p1837.2.d
	flavor = mfk_p1837.2.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	fire_only_once = yes

	option = { # Relief for the cotton growers: back Biddle's scheme to hold up the price
		name = mfk_p1837.2.a
		default_option = yes
		add_modifier = {
			name = mfk_cotton_relief_costs
			years = 2
			multiplier = money_amount_multiplier_medium
		}
		add_modifier = {
			name = mfk_biddle_cotton
			years = 2
		}
		every_subject_or_below = {
			limit = { is_involved_in_journal_entry = je_mfk_panic_1837 }
			add_modifier = {
				name = mfk_biddle_cotton
				years = 2
			}
		}
		je:je_mfk_panic_1837 ?= {
			set_variable = { name = mfk_biddle_months_var value = 0 }
		}
		custom_tooltip = mfk_panic_biddle_reckoning_tt
		if = {
			limit = { NOT = { has_variable = mfk_cq_shift_var } }
			set_variable = { name = mfk_cq_shift_var value = 0 }
		}
		change_variable = { name = mfk_cq_shift_var add = 0.25 }
		custom_tooltip = mfk_p1837.2.a_tt
		ai_chance = {
			base = 50
		}
	}

	option = { # The market will correct itself
		name = mfk_p1837.2.b
		ig:ig_landowners ?= {
			add_modifier = {
				name = delayed_emancipation
				days = normal_modifier_time
			}
		}
		set_global_variable = {
			name = mfk_cotton_collapse
			years = 3
		}
		custom_tooltip = mfk_p1837.2.b_tt
		ai_chance = {
			base = 50
		}
	}
}

# Biddle's Corner: the cotton scheme comes due (NAU)
mfk_p1837.6 = {
	type = country_event
	placement = root

	title = mfk_p1837.6.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = mfk_biddle_collapsed }
				desc = mfk_p1837.6.d_collapse
			}
			triggered_desc = {
				desc = mfk_p1837.6.d_holds
			}
		}
	}
	flavor = mfk_p1837.6.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	fire_only_once = yes

	immediate = {
		random_list = {
			60 = { set_variable = mfk_biddle_collapsed }
			40 = {}
		}
	}

	option = { # The scheme collapses
		name = mfk_p1837.6.a
		default_option = yes
		trigger = { has_variable = mfk_biddle_collapsed }
		je:je_mfk_panic_1837 ?= {
			add_progress = { value = -10 name = mfk_panic_1837_recovery_bar }
		}
		ig:ig_industrialists ?= {
			add_modifier = {
				name = mfk_biddle_ruined
				years = 2
			}
		}
	}

	option = { # The scheme holds
		name = mfk_p1837.6.b
		default_option = yes
		trigger = { NOT = { has_variable = mfk_biddle_collapsed } }
		je:je_mfk_panic_1837 ?= {
			add_progress = { value = 6 name = mfk_panic_1837_recovery_bar }
		}
	}
}
