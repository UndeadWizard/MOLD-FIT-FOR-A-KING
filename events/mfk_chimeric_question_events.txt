# The Division in the Commons: the decisive vote cuts a Chimeric Question bill short (NAU)
mfk_cq.10 = {
	type = country_event
	placement = root

	title = mfk_cq.10.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = mfk_cq_bill_compromise }
				desc = mfk_cq.10.d_compromise
			}
			triggered_desc = {
				desc = mfk_cq.10.d_enforce
			}
		}
	}
	flavor = mfk_cq.10.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	immediate = {
		je:je_mfk_chimeric_question ?= {
			save_scope_as = cq_journal_entry
		}
		set_variable = {
			name = mfk_cq_question_var
			value = "je:je_mfk_chimeric_question.scripted_bar_progress(mfk_cq_question_bar)"
		}
	}

	# The bill carries: the Question still backs it
	option = {
		name = mfk_cq.10.a
		default_option = yes
		trigger = {
			OR = {
				AND = {
					has_variable = mfk_cq_bill_compromise
					var:mfk_cq_question_var >= 50
				}
				AND = {
					has_variable = mfk_cq_bill_enforce
					var:mfk_cq_question_var < 50
				}
			}
		}
		mfk_cq_pass_bill_effect = yes
	}

	# Whip the vote: a narrow miss, forced through
	option = {
		name = mfk_cq.10.b
		trigger = {
			OR = {
				AND = {
					has_variable = mfk_cq_bill_compromise
					var:mfk_cq_question_var < 50
					var:mfk_cq_question_var >= 40
				}
				AND = {
					has_variable = mfk_cq_bill_enforce
					var:mfk_cq_question_var >= 50
					var:mfk_cq_question_var < 60
				}
			}
		}
		add_modifier = {
			name = mfk_cq_forced_settlement
			years = 2
		}
		mfk_cq_pass_bill_effect = yes
	}

	# The bill is defeated
	option = {
		name = mfk_cq.10.c
		trigger = {
			OR = {
				AND = {
					has_variable = mfk_cq_bill_compromise
					var:mfk_cq_question_var < 50
				}
				AND = {
					has_variable = mfk_cq_bill_enforce
					var:mfk_cq_question_var >= 50
				}
			}
		}
		cancel_enactment = yes
		je:je_mfk_chimeric_question ?= {
			add_progress = { value = 15 name = mfk_cq_tension_bar }
		}
		custom_tooltip = mfk_cq.10.c_tt
		mfk_cq_clear_bill_effect = yes
	}
}
